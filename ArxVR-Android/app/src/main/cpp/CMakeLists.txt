cmake_minimum_required(VERSION 3.22)
project(ArxVRAndroid LANGUAGES C CXX)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

get_filename_component(ARXVR_PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
get_filename_component(ARXVR_WORKSPACE_ROOT "${ARXVR_PROJECT_ROOT}/.." ABSOLUTE)

set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${ARXVR_WORKSPACE_ROOT}/SDL2" "${CMAKE_CURRENT_BINARY_DIR}/SDL2")

set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_ZLIB OFF CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
add_subdirectory(
    "${ARXVR_WORKSPACE_ROOT}/android-deps/third_party/freetype/src/main/jni/freetype"
    "${CMAKE_CURRENT_BINARY_DIR}/freetype"
)

set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
set(ALSOFT_BACKEND_OBOE OFF CACHE BOOL "" FORCE)
set(ALSOFT_BACKEND_OPENSL ON CACHE BOOL "" FORCE)
set(ALSOFT_BACKEND_WAVE OFF CACHE BOOL "" FORCE)
add_subdirectory(
    "${ARXVR_WORKSPACE_ROOT}/android-deps/third_party/openal/src/main/jni/openal"
    "${CMAKE_CURRENT_BINARY_DIR}/openal"
)

add_subdirectory(
    "${ARXVR_WORKSPACE_ROOT}/android-deps/app/src/main/jni/etcpaksources"
    "${CMAKE_CURRENT_BINARY_DIR}/etcpak"
)

set(SDL2_FOUND TRUE CACHE BOOL "" FORCE)
set(SDL2_INCLUDE_DIR "${ARXVR_WORKSPACE_ROOT}/SDL2/include" CACHE PATH "" FORCE)
set(SDL2_INCLUDE_DIRS "${ARXVR_WORKSPACE_ROOT}/SDL2/include" CACHE PATH "" FORCE)
set(SDL2_LIBRARY SDL2 CACHE STRING "" FORCE)
set(SDL2_LIBRARIES SDL2 CACHE STRING "" FORCE)

set(OPENAL_FOUND TRUE CACHE BOOL "" FORCE)
set(OPENAL_INCLUDE_DIR
    "${ARXVR_WORKSPACE_ROOT}/android-deps/third_party/openal/src/main/jni/openal/include"
    CACHE PATH "" FORCE)
set(OPENAL_EFX_INCLUDE_DIR "${OPENAL_INCLUDE_DIR}/AL" CACHE PATH "" FORCE)
set(OPENAL_LIBRARY OpenAL CACHE STRING "" FORCE)

set(FREETYPE_FOUND TRUE CACHE BOOL "" FORCE)
set(FREETYPE_INCLUDE_DIRS
    "${ARXVR_WORKSPACE_ROOT}/android-deps/third_party/freetype/src/main/jni/freetype/include"
    CACHE PATH "" FORCE)
set(FREETYPE_LIBRARY freetype CACHE STRING "" FORCE)
set(FREETYPE_LIBRARIES freetype CACHE STRING "" FORCE)

set(Boost_INCLUDE_DIRS "${ARXVR_WORKSPACE_ROOT}/android-deps/third_party/boost" CACHE PATH "" FORCE)
set(Boost_VERSION 109100 CACHE STRING "" FORCE)
set(Boost_VERSION_MACRO 109100 CACHE STRING "" FORCE)
set(GLM_INCLUDE_DIR "${ARXVR_WORKSPACE_ROOT}/android-deps/third_party/glm" CACHE PATH "" FORCE)
set(GLM_INCLUDE_DIRS "${ARXVR_WORKSPACE_ROOT}/android-deps/third_party/glm" CACHE PATH "" FORCE)
set(ETC_PAK_INCLUDE_DIR
    "${ARXVR_WORKSPACE_ROOT}/android-deps/app/src/main/jni/etcpaksources/etcpak"
    CACHE PATH "" FORCE)
include_directories("${ARXVR_WORKSPACE_ROOT}/android-deps/third_party/sse2neon")

set(USE_OPENGL ON CACHE BOOL "" FORCE)
set(USE_OPENAL ON CACHE BOOL "" FORCE)
set(WITH_SDL 2 CACHE STRING "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_CRASHREPORTER OFF CACHE BOOL "" FORCE)
set(BUILD_PROFILER OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(INSTALL_BLENDER_PLUGIN OFF CACHE BOOL "" FORCE)
set(BUILD_IO_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_LTO OFF CACHE BOOL "" FORCE)
set(SET_WARNING_FLAGS OFF CACHE BOOL "" FORCE)
set(USE_CXX17_FROM_CHARS_FLOAT ON CACHE BOOL "" FORCE)
set(CXX_STD_VERSION 2020 CACHE STRING "" FORCE)
set(default_DEBUG OFF)
set(DEBUG OFF CACHE BOOL "" FORCE)
set(DEBUG_GL OFF CACHE BOOL "" FORCE)

add_subdirectory("${ARXVR_WORKSPACE_ROOT}/ArxAndroid" "${CMAKE_CURRENT_BINARY_DIR}/arx")

target_link_libraries(arx android log EGL GLESv3 SDL2 OpenAL freetype etcpak)
target_compile_definitions(arx PRIVATE ARXVR_ANDROID_BUILD=1)
